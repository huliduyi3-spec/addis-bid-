import React, { useState } from 'react';
import { X, Wallet, CheckCircle2, Smartphone, Building, CreditCard } from 'lucide-react';
import { UserProfile, WalletTransaction, PaymentMethod } from '../types/auction';
import { formatETB } from '../utils/lubEngine';

interface WalletModalProps {
  user: UserProfile;
  currency: 'ETB' | 'USD';
  transactions: WalletTransaction[];
  onClose: () => void;
  onDeposit: (amount: number, method: PaymentMethod) => void;
}

export const WalletModal: React.FC<WalletModalProps> = ({
  user,
  transactions,
  onClose,
  onDeposit,
}) => {
  const [selectedMethod, setSelectedMethod] = useState<PaymentMethod>('telebirr');
  const [depositAmount, setDepositAmount] = useState<string>('500');
  const [phoneNumber, setPhoneNumber] = useState<string>('0911234567');
  const [receipt, setReceipt] = useState<boolean>(false);

  const handleDepositSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    const amt = parseFloat(depositAmount);
    if (isNaN(amt) || amt <= 0) return;

    onDeposit(amt, selectedMethod);
    setReceipt(true);
    setTimeout(() => setReceipt(false), 4000);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 bg-slate-950/85 backdrop-blur-md overflow-y-auto">
      <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-xl max-h-[92vh] overflow-y-auto shadow-2xl flex flex-col relative my-auto">
        <div className="flex items-center justify-between p-4 sm:p-6 border-b border-slate-800 sticky top-0 bg-slate-900/95 backdrop-blur z-20">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-amber-400/10 border border-amber-400/30 flex items-center justify-center text-amber-400">
              <Wallet className="w-5 h-5" />
            </div>
            <h2 className="font-display font-bold text-lg text-white">Wallet & Deposits</h2>
          </div>
          <button onClick={onClose} className="p-2 text-slate-400 hover:text-white rounded-lg">
            <X className="w-5 h-5" />
          </button>
        </div>

        <div className="p-6 space-y-6">
          <div className="p-4 bg-slate-950 border border-slate-800 rounded-xl">
            <div className="text-xs text-slate-400">Current Balance</div>
            <div className="text-3xl font-bold font-mono text-white mt-1">{formatETB(user.walletBalanceETB)}</div>
          </div>

          {receipt && (
            <div className="p-3 bg-emerald-950/40 border border-emerald-800/60 rounded-xl text-emerald-300 text-xs flex items-center gap-2">
              <CheckCircle2 className="w-4 h-4 text-emerald-400" />
              <span>Deposit completed successfully!</span>
            </div>
          )}

          <form onSubmit={handleDepositSubmit} className="space-y-4">
            <label className="block text-xs font-semibold text-slate-300">Deposit Amount (ETB)</label>
            <input
              type="number"
              value={depositAmount}
              onChange={(e) => setDepositAmount(e.target.value)}
              className="w-full px-3 py-2 bg-slate-950 border border-slate-700 rounded-lg text-sm text-white font-mono"
              required
            />
            <button
              type="submit"
              className="w-full py-3 bg-amber-400 hover:bg-amber-300 text-slate-950 font-bold text-xs rounded-lg"
            >
              Deposit via {selectedMethod}
            </button>
          </form>
        </div>
      </div>
    </div>
  );
};
