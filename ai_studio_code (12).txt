import React, { useState, useEffect } from 'react';
import { X, Trophy, ShieldCheck, AlertOctagon } from 'lucide-react';
import confetti from 'canvas-confetti';
import { AuctionItem } from '../types/auction';
import { analyzeLUB, formatETB } from '../utils/lubEngine';

interface AuctionResultModalProps {
  item: AuctionItem;
  currency: 'ETB' | 'USD';
  onClose: () => void;
  onClaimPrize?: () => void;
  isUserWinner?: boolean;
}

export const AuctionResultModal: React.FC<AuctionResultModalProps> = ({
  item,
  onClose,
  onClaimPrize,
  isUserWinner = false,
}) => {
  const analysis = analyzeLUB(item.bids);

  useEffect(() => {
    confetti({
      particleCount: 80,
      spread: 70,
      origin: { y: 0.6 },
    });
  }, []);

  const winningRecord = analysis.winningBidRecord || item.winningBid;
  const winningAmount = analysis.winningAmount || item.winningBid?.amount || 0;
  const savingsPercent = item.winningBid?.retailSavingsPercent || 99;

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 bg-slate-950/85 backdrop-blur-md overflow-y-auto">
      <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-4xl max-h-[92vh] overflow-y-auto shadow-2xl flex flex-col relative my-auto">
        <div className="flex items-center justify-between p-4 sm:p-6 border-b border-slate-800 sticky top-0 bg-slate-900/95 backdrop-blur z-20">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-amber-400/10 border border-amber-400/30 flex items-center justify-center text-amber-400">
              <Trophy className="w-5 h-5" />
            </div>
            <div>
              <div className="text-xs text-slate-400">OFFICIAL AUCTION RESULT</div>
              <h2 className="font-display font-bold text-lg text-white">{item.title}</h2>
            </div>
          </div>
          <button onClick={onClose} className="p-2 text-slate-400 hover:text-white rounded-lg">
            <X className="w-5 h-5" />
          </button>
        </div>

        <div className="p-6 bg-slate-950/60 border-b border-slate-800">
          <div className="flex flex-col sm:flex-row items-center justify-between gap-4">
            <div>
              <div className="text-xs font-mono text-amber-400 font-semibold">Winning Bidder</div>
              <h3 className="text-2xl font-display font-bold text-white mt-1">
                {winningRecord?.userName || 'Dawit M.'} claims the prize!
              </h3>
            </div>
            <div className="text-right">
              <div className="text-xs text-slate-400">Winning Price</div>
              <div className="text-3xl font-mono font-extrabold text-amber-400">{formatETB(winningAmount)}</div>
              <div className="text-xs text-emerald-400 font-semibold mt-1">Saved {savingsPercent}% off retail!</div>
              {isUserWinner && onClaimPrize && (
                <button
                  onClick={onClaimPrize}
                  className="mt-3 py-2 px-4 bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-bold text-xs rounded-lg"
                >
                  Claim Prize & Delivery
                </button>
              )}
            </div>
          </div>
        </div>

        <div className="p-4 sm:p-6 space-y-4">
          <h4 className="font-display font-bold text-sm text-white">Full Bid Verification Table</h4>
          <div className="bg-slate-950 border border-slate-800 rounded-xl overflow-x-auto max-h-72">
            <table className="w-full text-left text-xs font-mono">
              <thead className="bg-slate-900 text-slate-400 border-b border-slate-800">
                <tr>
                  <th className="py-2 px-4">Amount</th>
                  <th className="py-2 px-4">Bidders</th>
                  <th className="py-2 px-4">Result</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-800/60">
                {analysis.frequencies.map((freq) => (
                  <tr key={freq.amount} className={freq.isWinning ? 'bg-amber-400/10' : ''}>
                    <td className="py-2 px-4 font-bold text-white">{formatETB(freq.amount)}</td>
                    <td className="py-2 px-4 text-slate-300">{freq.bidders.join(', ')}</td>
                    <td className="py-2 px-4">
                      {freq.isWinning ? (
                        <span className="text-amber-400 font-bold">WINNER (Lowest & Unique)</span>
                      ) : freq.isUnique ? (
                        <span className="text-slate-400">Unique (Higher)</span>
                      ) : (
                        <span className="text-rose-400">Duplicate Cancelled</span>
                      )}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      </div>
    </div>
  );
};
