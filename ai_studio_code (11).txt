import React, { useState } from 'react';
import { X, ShieldCheck, CheckCircle2 } from 'lucide-react';
import { AuctionItem, UserBid, UserProfile } from '../types/auction';
import { formatETB, formatUSD, formatTimeRemaining } from '../utils/lubEngine';

interface AuctionDetailModalProps {
  item: AuctionItem;
  user: UserProfile;
  currency: 'ETB' | 'USD';
  userBidsOnItem: UserBid[];
  onClose: () => void;
  onPlaceBid: (auctionId: string, amount: number) => boolean;
  onPlaceSpreadBids: (auctionId: string, amounts: number[]) => void;
  onOpenWallet: () => void;
}

export const AuctionDetailModal: React.FC<AuctionDetailModalProps> = ({
  item,
  user,
  currency,
  userBidsOnItem,
  onClose,
  onPlaceBid,
  onPlaceSpreadBids,
  onOpenWallet,
}) => {
  const [bidAmountInput, setBidAmountInput] = useState<string>('2.15');
  const [errorMsg, setErrorMsg] = useState<string | null>(null);
  const [successMsg, setSuccessMsg] = useState<string | null>(null);

  const timeLeft = Math.max(0, item.endTime - Date.now());
  const timer = formatTimeRemaining(timeLeft);

  const handleSingleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setErrorMsg(null);
    setSuccessMsg(null);

    const val = parseFloat(bidAmountInput);
    if (isNaN(val) || val <= 0) {
      setErrorMsg('Please enter a valid positive bid amount.');
      return;
    }

    if (val > item.maxBidLimitETB) {
      setErrorMsg(`Maximum allowed bid for this item is ${formatETB(item.maxBidLimitETB)}.`);
      return;
    }

    if (user.walletBalanceETB < item.entryFeeETB) {
      setErrorMsg(`Insufficient wallet balance. You need ${formatETB(item.entryFeeETB)}.`);
      return;
    }

    const success = onPlaceBid(item.id, Number(val.toFixed(2)));
    if (success) {
      setSuccessMsg(`Secret bid of ${formatETB(val)} placed successfully!`);
      setBidAmountInput((val + 0.37).toFixed(2));
      setTimeout(() => setSuccessMsg(null), 4000);
    }
  };

  const handleSpreadSubmit = () => {
    setErrorMsg(null);
    setSuccessMsg(null);
    const feeNeeded = item.entryFeeETB * 3;
    if (user.walletBalanceETB < feeNeeded) {
      setErrorMsg(`Placing 3 spread bids requires ${formatETB(feeNeeded)}.`);
      return;
    }

    const base = parseFloat(bidAmountInput) || 2.5;
    const b1 = Number((Math.max(0.12, base * 0.7 + Math.random() * 0.4)).toFixed(2));
    const b2 = Number((base + Math.random() * 0.8).toFixed(2));
    const b3 = Number((base * 1.5 + Math.random() * 0.9).toFixed(2));

    onPlaceSpreadBids(item.id, [b1, b2, b3]);
    setSuccessMsg(`3 Strategic Spread Bids placed (${formatETB(b1)}, ${formatETB(b2)}, ${formatETB(b3)})!`);
    setTimeout(() => setSuccessMsg(null), 4500);
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 bg-slate-950/80 backdrop-blur-md overflow-y-auto">
      <div className="bg-slate-900 border border-slate-800 rounded-2xl w-full max-w-4xl max-h-[92vh] overflow-y-auto shadow-2xl flex flex-col relative my-auto">
        <div className="flex items-center justify-between p-4 sm:p-6 border-b border-slate-800 sticky top-0 bg-slate-900/95 backdrop-blur z-20">
          <div>
            <div className="flex items-center gap-2 text-xs text-slate-400">
              <span className="text-amber-400 font-semibold uppercase">{item.category}</span>
              <span aria-hidden="true">·</span>
              <span>Lot #{item.id.replace('auc-', '').toUpperCase()}</span>
            </div>
            <h2 className="font-display font-bold text-lg sm:text-xl text-white tracking-tight mt-0.5">
              {item.title}
            </h2>
          </div>
          <button onClick={onClose} className="p-2 text-slate-400 hover:text-white rounded-lg">
            <X className="w-5 h-5" />
          </button>
        </div>

        <div className="p-4 sm:p-6 grid grid-cols-1 lg:grid-cols-12 gap-8">
          <div className="lg:col-span-7 flex flex-col space-y-6">
            <div className="relative aspect-[4/3] rounded-xl overflow-hidden bg-slate-950 border border-slate-800">
              <img src={item.image} alt={item.title} className="w-full h-full object-cover" />
              <div className="absolute top-3 left-3 bg-slate-950/90 border border-slate-800 px-3 py-1.5 rounded-lg text-xs text-white">
                <span className="text-slate-400 mr-2">Market Value:</span>
                <span className="font-mono font-bold text-amber-300">
                  {currency === 'ETB' ? formatETB(item.retailValueETB) : formatUSD(item.retailValueUSD)}
                </span>
              </div>
            </div>

            <div>
              <h4 className="text-xs font-semibold text-slate-300 uppercase tracking-wider mb-2">
                Item Description & Features
              </h4>
              <p className="text-xs sm:text-sm text-slate-400 leading-relaxed">{item.description}</p>
            </div>

            <div className="p-4 bg-slate-950 border border-slate-800 rounded-xl space-y-2 text-xs">
              <div className="font-semibold text-white">Technical Specifications</div>
              <ul className="space-y-1 text-slate-400 list-disc list-inside">
                {item.specs.map((sp, idx) => (
                  <li key={idx}>{sp}</li>
                ))}
              </ul>
            </div>
          </div>

          <div className="lg:col-span-5 flex flex-col space-y-6">
            <div className="bg-slate-950 border border-slate-800 rounded-xl p-4 space-y-3">
              <div className="flex items-center justify-between text-xs">
                <span className="text-slate-400">Auction Status:</span>
                <span className="text-emerald-400 font-semibold font-mono">
                  {timer.hours}:{timer.minutes}:{timer.seconds} remaining
                </span>
              </div>
              <div className="flex items-center justify-between text-xs">
                <span className="text-slate-400">Entry Fee per Bid:</span>
                <span className="font-mono font-bold text-amber-400">{formatETB(item.entryFeeETB)}</span>
              </div>
              <div className="flex items-center justify-between text-xs pt-2 border-t border-slate-800">
                <span className="text-slate-400">Your Wallet Balance:</span>
                <span className="font-mono font-bold text-white">{formatETB(user.walletBalanceETB)}</span>
              </div>
            </div>

            <form onSubmit={handleSingleSubmit} className="space-y-4">
              <div>
                <label className="block text-xs font-semibold text-slate-300 mb-1.5">
                  Enter Your Secret Bid (ETB)
                </label>
                <div className="relative">
                  <span className="absolute left-3 top-1/2 -translate-y-1/2 text-xs font-mono font-bold text-amber-400">
                    ETB
                  </span>
                  <input
                    type="number"
                    step="0.01"
                    min="0.01"
                    max={item.maxBidLimitETB}
                    value={bidAmountInput}
                    onChange={(e) => setBidAmountInput(e.target.value)}
                    className="w-full pl-14 pr-4 py-2.5 bg-slate-950 border border-slate-700 rounded-lg font-mono text-base font-bold text-white focus:outline-none focus:border-amber-400"
                    required
                  />
                </div>
              </div>

              {errorMsg && <div className="text-xs text-rose-400 p-2.5 bg-rose-950/30 rounded-lg">{errorMsg}</div>}
              {successMsg && <div className="text-xs text-emerald-400 p-2.5 bg-emerald-950/30 rounded-lg">{successMsg}</div>}

              <button
                type="submit"
                className="w-full py-3 bg-amber-400 hover:bg-amber-300 text-slate-950 font-bold text-xs rounded-lg transition-all"
              >
                Place Sealed Secret Bid ({formatETB(item.entryFeeETB)} fee)
              </button>

              <button
                type="button"
                onClick={handleSpreadSubmit}
                className="w-full py-2.5 bg-slate-800 hover:bg-slate-700 text-slate-200 font-semibold text-xs rounded-lg transition-colors border border-slate-700"
              >
                Place 3x Strategic Spread Bids ({formatETB(item.entryFeeETB * 3)})
              </button>
            </form>
          </div>
        </div>
      </div>
    </div>
  );
};
